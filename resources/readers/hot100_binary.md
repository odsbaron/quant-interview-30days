### 快速理解

**题干**

**触发信号**

**代码阅读顺序**

**复杂度（学习版估计）**
- 通常为 O(n) 级别（单次线性扫描），空间多为 O(1)~O(n)。


```python
# ==============================
# 母题：33. 搜索旋转排序数组
# 这段代码分两部分：
# 1) 上半部分是算法函数；
# 2) 下半部分是样例与断言，便于你直接运行验证。
# ==============================

# 函数入口：search。先关注输入参数，再看返回值。
def search(nums, target):
    l, r = 0, len(nums) - 1
    # 当条件成立时持续推进，直到边界条件触发退出。
    while l <= r:
        m = (l + r) // 2
        # 分支判断：不同条件走不同处理路径。
        if nums[m] == target:
            # 返回当前函数结果。
            return m
        # 分支判断：不同条件走不同处理路径。
        if nums[l] <= nums[m]:
            # 分支判断：不同条件走不同处理路径。
            if nums[l] <= target < nums[m]:
                r = m - 1
            # 兜底分支：处理其余情况。
            else:
                l = m + 1
        # 兜底分支：处理其余情况。
        else:
            # 分支判断：不同条件走不同处理路径。
            if nums[m] < target <= nums[r]:
                l = m + 1
            # 兜底分支：处理其余情况。
            else:
                r = m - 1
    # 返回当前函数结果。
    return -1

# 准备测试样例：每组包含输入和期望输出。
cases = [
    (([4,5,6,7,0,1,2], 0), 4),
    (([4,5,6,7,0,1,2], 3), -1),
    (([1], 0), -1),
]
# 逐个运行样例，观察输出并与预期对比。

    out = search(nums, target)
    show_case("Search in Rotated Sorted Array", {"nums": nums, "target": target}, out, expected)
    # 断言校验：若失败会抛异常，说明逻辑需要排查。
    assert out == expected
```

```python
# 母题：74. 搜索二维矩阵

# 函数入口：search_matrix。先关注输入参数，再看返回值。
def search_matrix(matrix, target):
    m, n = len(matrix), len(matrix[0])
    l, r = 0, m * n - 1
    # 当条件成立时持续推进，直到边界条件触发退出。
    while l <= r:
        mid = (l + r) // 2
        x = matrix[mid // n][mid % n]
        # 分支判断：不同条件走不同处理路径。
        if x == target:
            # 返回当前函数结果。
            return True
        # 分支判断：不同条件走不同处理路径。
        if x < target:
            l = mid + 1
        # 兜底分支：处理其余情况。
        else:

    return False

cases = [
```
