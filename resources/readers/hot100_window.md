```python
# 函数入口：length_of_longest_substring。先关注输入参数，再看返回值。
def length_of_longest_substring(s):
    last = {}
    l = 0
    ans = 0
    # 循环推进状态：每一步都在逼近最终答案。
    for r, ch in enumerate(s):
        # 分支判断：不同条件走不同处理路径。
```

```python
# ==============================
# 母题：438. 找到字符串中所有字母异位词
# 这段代码分两部分：
# 1) 上半部分是算法函数；
# 2) 下半部分是样例与断言，便于你直接运行验证。
# ==============================

from collections import Counter

# 函数入口：find_anagrams。先关注输入参数，再看返回值。
def find_anagrams(s, p):
    need = Counter(p)
    win = Counter()
    k = len(p)
    ans = []
    # 循环推进状态：每一步都在逼近最终答案。
    for i, ch in enumerate(s):
        win[ch] += 1
        # 分支判断：不同条件走不同处理路径。
        if i >= k:
            left = s[i - k]
            win[left] -= 1
            # 分支判断：不同条件走不同处理路径。
            if win[left] == 0:
                del win[left]
        # 分支判断：不同条件走不同处理路径。
        if i >= k - 1 and win == need:
            ans.append(i - k + 1)
    # 返回当前函数结果。
    return ans

# 准备测试样例：每组包含输入和期望输出。
cases = [
    (("cbaebabacd", "abc"), [0, 6]),
    (("abab", "ab"), [0, 1, 2]),
]
# 逐个运行样例，观察输出并与预期对比。

    out = find_anagrams(s, p)
    show_case("Find Anagrams", {"s": s, "p": p}, out, expected)
    # 断言校验：若失败会抛异常，说明逻辑需要排查。
```

- 通常为 O(V+E) 或 O(m*n)（图/网格 BFS），空间与队列规模相关。


```python
# ==============================
# 母题：239. 滑动窗口最大值
# 这段代码分两部分：
# 1) 上半部分是算法函数；
# 2) 下半部分是样例与断言，便于你直接运行验证。
# ==============================

from collections import deque

# 函数入口：max_sliding_window。先关注输入参数，再看返回值。
def max_sliding_window(nums, k):
    q = deque()
    ans = []
    # 循环推进状态：每一步都在逼近最终答案。
    for i, x in enumerate(nums):
        # 当条件成立时持续推进，直到边界条件触发退出。
        while q and nums[q[-1]] <= x:
            q.pop()
        q.append(i)
        # 分支判断：不同条件走不同处理路径。
        if q[0] <= i - k:
            q.popleft()
        # 分支判断：不同条件走不同处理路径。
        if i >= k - 1:
            ans.append(nums[q[0]])
    # 返回当前函数结果。
    return ans

# 准备测试样例：每组包含输入和期望输出。
cases = [
    (([1,3,-1,-3,5,3,6,7], 3), [3,3,5,5,6,7]),
    (([1], 1), [1]),
]
# 逐个运行样例，观察输出并与预期对比。
for (nums, k), expected in cases:
    out = max_sliding_window(nums, k)
    show_case("Sliding Window Maximum", {"nums": nums, "k": k}, out, expected)
    # 断言校验：若失败会抛异常，说明逻辑需要排查。
    assert out == expected
# 输出提示，帮助你确认该题样例已通过。
```

**复杂度（学习版估计）**

```python
# ==============================
# 母题：76. 最小覆盖子串
# 这段代码分两部分：
# 1) 上半部分是算法函数；
# 2) 下半部分是样例与断言，便于你直接运行验证。
# ==============================

from collections import Counter

# 函数入口：min_window。先关注输入参数，再看返回值。
def min_window(s, t):
    need = Counter(t)
    missing = len(t)
    l = 0
    best_len = float("inf")
    best_l = 0
    # 循环推进状态：每一步都在逼近最终答案。
    for r, ch in enumerate(s):
        # 分支判断：不同条件走不同处理路径。
        if need[ch] > 0:
            missing -= 1
        need[ch] -= 1
        # 当条件成立时持续推进，直到边界条件触发退出。
        while missing == 0:
            # 分支判断：不同条件走不同处理路径。
            if r - l + 1 < best_len:
                best_len = r - l + 1
                best_l = l
            left = s[l]
            need[left] += 1
            # 分支判断：不同条件走不同处理路径。
            if need[left] > 0:
                missing += 1
            l += 1
    # 返回当前函数结果。
    return "" if best_len == float("inf") else s[best_l: best_l + best_len]

# 准备测试样例：每组包含输入和期望输出。
cases = [
    (("ADOBECODEBANC", "ABC"), "BANC"),
    (("a", "a"), "a"),
    (("a", "aa"), ""),
]
# 逐个运行样例，观察输出并与预期对比。
for (s, t), expected in cases:
    out = min_window(s, t)
    show_case("Minimum Window Substring", {"s": s, "t": t}, out, expected)
    # 断言校验：若失败会抛异常，说明逻辑需要排查。
    assert out == expected
# 输出提示，帮助你确认该题样例已通过。
print("母题通过")
```

```python
# ==============================
# 母题：209. 长度最小的子数组（正数）
# 这段代码分两部分：
# 1) 上半部分是算法函数；
# 2) 下半部分是样例与断言，便于你直接运行验证。
# ==============================

# 函数入口：min_sub_array_len。先关注输入参数，再看返回值。
def min_sub_array_len(target, nums):
    l = 0
    s = 0
    ans = float("inf")
    # 循环推进状态：每一步都在逼近最终答案。
    for r, x in enumerate(nums):
        s += x
        # 当条件成立时持续推进，直到边界条件触发退出。
        while s >= target:
            ans = min(ans, r - l + 1)
            s -= nums[l]
            l += 1
    # 返回当前函数结果。
    return 0 if ans == float("inf") else ans

# 准备测试样例：每组包含输入和期望输出。

    ((7, [2,3,1,2,4,3]), 2),

]
# 逐个运行样例，观察输出并与预期对比。

    out = min_sub_array_len(target, nums)
    show_case("Minimum Size Subarray Sum", {"target": target, "nums": nums}, out, expected)
    # 断言校验：若失败会抛异常，说明逻辑需要排查。
```
