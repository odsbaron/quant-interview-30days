```python
# ==============================
# 母题：20. 有效的括号
# 这段代码分两部分：

# 函数入口：is_valid。先关注输入参数，再看返回值。
def is_valid(s):
    mp = {')': '(', ']': '[', '}': '{'}
    st = []
    # 循环推进状态：每一步都在逼近最终答案。
    for ch in s:
        # 分支判断：不同条件走不同处理路径。
        if ch in mp.values():
            st.append(ch)
        # 兜底分支：处理其余情况。
        else:
            # 分支判断：不同条件走不同处理路径。
            if not st or st[-1] != mp[ch]:
                # 返回当前函数结果。
                return False
            st.pop()
    # 返回当前函数结果。
    return not st

# 准备测试样例：每组包含输入和期望输出。
cases = [("()", True), ("()[]{}", True), ("(]", False), ("([)]", False)]
# 逐个运行样例，观察输出并与预期对比。
```

## 母题：155. 最小栈

**复杂度（学习版估计）**

```python
# ==============================
# 母题：155. 最小栈
# 这段代码分两部分：
# 1) 上半部分是算法函数；
# 2) 下半部分是样例与断言，便于你直接运行验证。
# ==============================

# 定义数据结构类（封装状态与操作）。
class MinStack:
    # 函数入口：__init__。先关注输入参数，再看返回值。
    def __init__(self):
        self.st = []
        self.mn = []
    # 函数入口：push。先关注输入参数，再看返回值。
    def push(self, val):
        self.st.append(val)
        # 分支判断：不同条件走不同处理路径。
        if not self.mn or val <= self.mn[-1]:
            self.mn.append(val)
    # 函数入口：pop。先关注输入参数，再看返回值。
    def pop(self):
        v = self.st.pop()
        # 分支判断：不同条件走不同处理路径。
        if v == self.mn[-1]:
            self.mn.pop()
    # 函数入口：top。先关注输入参数，再看返回值。
    def top(self):
        # 返回当前函数结果。
        return self.st[-1]
    # 函数入口：getMin。先关注输入参数，再看返回值。
    def getMin(self):
        # 返回当前函数结果。
        return self.mn[-1]

ms = MinStack()
ms.push(-2)
ms.push(0)
ms.push(-3)
out1 = ms.getMin()
ms.pop()
out2 = ms.top()
out3 = ms.getMin()
show_case("MinStack", "ops", (out1, out2, out3), (-3, 0, -2))
# 断言校验：若失败会抛异常，说明逻辑需要排查。
assert (out1, out2, out3) == (-3, 0, -2)
# 输出提示，帮助你确认该题样例已通过。
print("母题通过")
```

```python
# ==============================
# 母题：394. 字符串解码
# 这段代码分两部分：

# ==============================

# 函数入口：decode_string。先关注输入参数，再看返回值。
def decode_string(s):
    num = 0
    cur = ""
    st = []
    # 循环推进状态：每一步都在逼近最终答案。
    for ch in s:
        # 分支判断：不同条件走不同处理路径。
        if ch.isdigit():
            num = num * 10 + int(ch)
        # 备用分支：处理前一个条件不满足的情况。
        elif ch == '[':
            st.append((cur, num))
            cur = ""
            num = 0
        # 备用分支：处理前一个条件不满足的情况。
        elif ch == ']':
            prev, k = st.pop()
            cur = prev + cur * k
        # 兜底分支：处理其余情况。
        else:
            cur += ch
    # 返回当前函数结果。
    return cur

# 准备测试样例：每组包含输入和期望输出。
cases = [("3[a]2[bc]", "aaabcbc"), ("3[a2[c]]", "accaccacc"), ("2[abc]3[cd]ef", "abcabccdcdcdef")]
```

**复杂度（学习版估计）**

```python
# ==============================
# 母题：739. 每日温度
# 这段代码分两部分：
# 1) 上半部分是算法函数；
# 2) 下半部分是样例与断言，便于你直接运行验证。
# ==============================

# 函数入口：daily_temperatures。先关注输入参数，再看返回值。
def daily_temperatures(temperatures):
    n = len(temperatures)
    ans = [0] * n
    st = []
    # 循环推进状态：每一步都在逼近最终答案。
    for i, t in enumerate(temperatures):
        # 当条件成立时持续推进，直到边界条件触发退出。
        while st and temperatures[st[-1]] < t:
            j = st.pop()
            ans[j] = i - j
        st.append(i)
    # 返回当前函数结果。
    return ans

inp = [73,74,75,71,69,72,76,73]
expected = [1,1,4,2,1,1,0,0]
out = daily_temperatures(inp)
show_case("Daily Temperatures", {"temperatures": inp}, out, expected)
# 断言校验：若失败会抛异常，说明逻辑需要排查。

# 输出提示，帮助你确认该题样例已通过。
```

```python
# ==============================
# 母题：215. 数组中的第 K 个最大元素
# 这段代码分两部分：
# 1) 上半部分是算法函数；
# 2) 下半部分是样例与断言，便于你直接运行验证。
# ==============================

import heapq

# 函数入口：find_kth_largest。先关注输入参数，再看返回值。
def find_kth_largest(nums, k):
    # 返回当前函数结果。
```
